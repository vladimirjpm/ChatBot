Что запущено прямо сейчас

# Инфраструктура (опционально, нужна на этапе 4)
docker compose up -d
# Frontend — первый раз (установка зависимостей)
cmd /c "cd C:\_Proj\ChatBot\chatbot-ui && npm install"
# Frontend — запуск
cmd /c "cd C:\_Proj\ChatBot\chatbot-ui && npm run dev"
→ http://localhost:5173
# Backend API
cd C:\_Proj\ChatBot\ChatBot.Api
dotnet run
→ http://localhost:5249/openapi/v1.json
